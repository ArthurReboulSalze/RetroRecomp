# Appends the hash of a generator script, or the supplied generator stamp.
function(rr_tool_hash RR_VAR RR_TOOL)
    set(RR_TOOL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/${RR_TOOL}")
    if(EXISTS "${RR_TOOL_PATH}")
        file(SHA256 "${RR_TOOL_PATH}" RR_TOOL_HASH)
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${RR_TOOL_PATH}")
    elseif(RR_X68000_TOOL_STAMP)
        set(RR_TOOL_HASH "stamp:${RR_X68000_TOOL_STAMP}")
    else()
        message(FATAL_ERROR "Missing tools/${RR_TOOL}; pass RR_X68000_TOOL_STAMP from the converter")
    endif()
    set(${RR_VAR} "${${RR_VAR}}|${RR_TOOL_HASH}" PARENT_SCOPE)
endfunction()

    ../x68000_machine/base74/cpu_bus.c ../x68000_machine/base74/audio_mix.c)

    list(APPEND RR_STATE_FILES ../x68000_machine/state/snapshot.c ../x68000_machine/state/rr_state.c)

    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${RR_STATE_SOURCE}" RR_STATE_HASH)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${RR_STATE_SOURCE}")
    string(APPEND RR_STATE_ABI_INPUT "|${RR_STATE_HASH}")
endforeach()

if(RR_X68000_MIT_GLUE)
    # States of the two glue builds are never interchangeable.

    rr_tool_hash(RR_STATE_ABI_INPUT x68000_state_hardware.py)
endif()

    file(SHA256 "${RR_X68000_GENERATED}/m68kcpu.c" RR_STATE_HASH)
    string(APPEND RR_STATE_ABI_INPUT "|native|${RR_STATE_HASH}")

    target_compile_options(rr_px68k_reference PRIVATE -O2 -fno-fast-math
