cmake_minimum_required(VERSION 3.20)
project(RetroRecompAmiga LANGUAGES C CXX)

# The Amiga player (MIT, ours) on vAmiga's Core (MPL-2.0), adapted at build time by retrorecomp/amiga_vamiga.py.
#   RR_VAMIGA_CORE        the adapted Core folder (required)
#   RR_AMIGA_GENERATED    the generated native steps (steps/*.cpp, steps/index.c, step_semantics.h): a native build
#   RR_AMIGA_PAYLOAD_DIR  the embedded install and resources (amiga_payload.h, game_resources.rc): an exported game
#   RR_AMIGA_STATE_INPUT  what else the quick-state key depends on (the converter's stamp)
set(RR_VAMIGA_CORE "" CACHE PATH "Adapted vAmiga Core folder")
set(RR_AMIGA_GENERATED "" CACHE PATH "Generated native steps of one game")
set(RR_AMIGA_PAYLOAD_DIR "" CACHE PATH "Private per-game resources; never a public release input")
set(RR_AMIGA_STATE_INPUT "" CACHE STRING "Converter stamp entering the quick-state key")
option(RR_AMIGA_TOOLS "Also build the opcode-table tool of the conversion" OFF)
if(NOT RR_VAMIGA_CORE OR NOT EXISTS "${RR_VAMIGA_CORE}/CMakeLists.txt")
    message(FATAL_ERROR "RR_VAMIGA_CORE must name the adapted vAmiga Core folder")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory("${RR_VAMIGA_CORE}" vacore)
# vAmiga's own warnings-as-errors stay on its own files; they are not passed on to ours (its libraries publish them).
foreach(RR_VAMIGA_TARGET VACore rvlib utlib)
    if(TARGET ${RR_VAMIGA_TARGET})
        set_property(TARGET ${RR_VAMIGA_TARGET} PROPERTY INTERFACE_COMPILE_OPTIONS "")
    endif()
endforeach()

get_filename_component(RR_COMMON_NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(RR_WHDLOAD "${RR_COMMON_NATIVE}/amiga_whdload")

add_library(rr_amiga_session STATIC amiga_session.cpp "${RR_WHDLOAD}/rr_resload.c" "${RR_WHDLOAD}/rr_decrunch.c")
target_include_directories(rr_amiga_session PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${RR_WHDLOAD}")
target_link_libraries(rr_amiga_session PUBLIC VACore)
if(WIN32)
    target_link_libraries(rr_amiga_session PUBLIC ws2_32)
endif()

# The quick-state key: states of different builds are never mixed.
set(RR_STATE_INPUT "amiga-state-v1|${CMAKE_SYSTEM_NAME}|${CMAKE_SYSTEM_PROCESSOR}|${CMAKE_SIZEOF_VOID_P}|${CMAKE_CXX_COMPILER_ID}|${RR_AMIGA_STATE_INPUT}")
foreach(RR_STATE_SOURCE amiga_session.cpp amiga_session.h "${RR_WHDLOAD}/rr_resload.c" "${RR_WHDLOAD}/rr_resload.h" "${RR_VAMIGA_CORE}/../ready.json")
    if(NOT IS_ABSOLUTE "${RR_STATE_SOURCE}")
        set(RR_STATE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/${RR_STATE_SOURCE}")
    endif()
    file(SHA256 "${RR_STATE_SOURCE}" RR_STATE_HASH)
    string(APPEND RR_STATE_INPUT "|${RR_STATE_HASH}")
endforeach()

add_executable(amiga_player amiga_player.c
    "${RR_COMMON_NATIVE}/retro_menu.c" "${RR_COMMON_NATIVE}/platform_io.c")
foreach(RR_KIT_FILE audio events input main menu options pointer ring timing video)
    target_sources(amiga_player PRIVATE "${RR_COMMON_NATIVE}/hostkit/shell/rrhk_${RR_KIT_FILE}.c")
endforeach()
target_include_directories(amiga_player PRIVATE "${RR_COMMON_NATIVE}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(amiga_player PRIVATE rr_amiga_session)
set_target_properties(amiga_player PROPERTIES LINKER_LANGUAGE CXX OUTPUT_NAME "RetroRecomp-Amiga")
if(NOT RR_RECOMP_VERSION)
    set(RR_RECOMP_VERSION "dev")
endif()
target_compile_definitions(amiga_player PRIVATE RR_RECOMP_VERSION="${RR_RECOMP_VERSION}" SDL_MAIN_HANDLED)

if(RR_AMIGA_GENERATED)
    file(GLOB RR_AMIGA_STEPS "${RR_AMIGA_GENERATED}/steps/*.cpp" "${RR_AMIGA_GENERATED}/steps/*.c")
    if(NOT RR_AMIGA_STEPS)
        message(FATAL_ERROR "RR_AMIGA_GENERATED holds no steps")
    endif()
    target_sources(rr_amiga_session PRIVATE ${RR_AMIGA_STEPS})
    target_include_directories(rr_amiga_session PRIVATE "${RR_AMIGA_GENERATED}")
    target_compile_definitions(rr_amiga_session PRIVATE RR_AMIGA_NATIVE=1)
    file(SHA256 "${RR_AMIGA_GENERATED}/steps/index.c" RR_STATE_HASH)
    string(APPEND RR_STATE_INPUT "|native|${RR_STATE_HASH}")
endif()
string(SHA256 RR_STATE_ABI "${RR_STATE_INPUT}")
string(SUBSTRING "${RR_STATE_ABI}" 0 64 RR_STATE_ABI)
target_compile_definitions(amiga_player PRIVATE RR_AMIGA_STATE_ABI="${RR_STATE_ABI}")

if(RR_AMIGA_PAYLOAD_DIR)
    target_include_directories(amiga_player PRIVATE "${RR_AMIGA_PAYLOAD_DIR}")
    target_compile_definitions(amiga_player PRIVATE RR_AMIGA_LINKED)
    if(WIN32)
        # No console window for the game; the headless validation entry stays.
        if(MSVC)
            set_target_properties(amiga_player PROPERTIES WIN32_EXECUTABLE TRUE)
            target_link_options(amiga_player PRIVATE /ENTRY:mainCRTStartup)
        else()
            target_link_options(amiga_player PRIVATE -Wl,--subsystem,windows -Wl,--entry,mainCRTStartup)
        endif()
        enable_language(RC)
        target_sources(amiga_player PRIVATE "${RR_AMIGA_PAYLOAD_DIR}/game_resources.rc" "${RR_AMIGA_PAYLOAD_DIR}/game_metadata.rc")
    else()
        enable_language(ASM)
        target_sources(amiga_player PRIVATE "${RR_AMIGA_PAYLOAD_DIR}/embedded_resources.S" "${RR_AMIGA_PAYLOAD_DIR}/embedded_resources.c")
    endif()
endif()

find_package(SDL2 CONFIG REQUIRED)
if(TARGET SDL2::SDL2-static)
    target_link_libraries(amiga_player PRIVATE SDL2::SDL2-static)
else()
    target_link_libraries(amiga_player PRIVATE SDL2::SDL2)
endif()
if(MSVC)
    target_compile_definitions(amiga_player PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(amiga_player PRIVATE /O2 /utf-8)
    target_link_libraries(amiga_player PRIVATE gdi32 shell32)
else()
    target_compile_options(amiga_player PRIVATE -O2 -fno-fast-math)
    target_link_libraries(amiga_player PRIVATE m)
    if(WIN32)
        target_link_libraries(amiga_player PRIVATE gdi32 shell32 user32)
    endif()
endif()

if(RR_AMIGA_TOOLS)
    add_executable(amiga_moira_map moira_map.cpp)
    target_link_libraries(amiga_moira_map PRIVATE VACore)
    if(WIN32)
        target_link_libraries(amiga_moira_map PRIVATE ws2_32)
    endif()
endif()
