cmake_minimum_required(VERSION 3.16)
project(retro_credits LANGUAGES C)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
set(BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(BUILD_LITE ON CACHE BOOL "" FORCE)
set(LIBXMP_PIC ON CACHE BOOL "" FORCE)
set(LIBXMP_DISABLE_IT ON CACHE BOOL "" FORCE)
add_subdirectory("${XMP_SOURCE}" xmp EXCLUDE_FROM_ALL)
add_library(retro_credits SHARED player.c)
target_include_directories(retro_credits PRIVATE "${MINIAUDIO_INCLUDE}")
target_link_libraries(retro_credits PRIVATE xmp_lite_static)
set_target_properties(retro_credits PROPERTIES C_STANDARD 99)
if(NOT MSVC)
  target_compile_options(retro_credits PRIVATE -O2 -fvisibility=hidden)
  target_link_libraries(retro_credits PRIVATE m dl pthread)
  target_link_options(retro_credits PRIVATE -Wl,--exclude-libs,ALL -s)
endif()
